cmake_minimum_required(VERSION 3.22)
project(ddp_forge VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(DDP_SANITIZE "AddressSanitizer + UBSan" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wshadow)
if(DDP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(ddp INTERFACE)
target_include_directories(ddp INTERFACE include)
target_link_libraries(ddp INTERFACE Eigen3::Eigen)
add_executable(ddp_experiments apps/ddp_experiments.cpp)
target_link_libraries(ddp_experiments PRIVATE ddp)
enable_testing()
find_package(GTest REQUIRED)
add_executable(ddp_tests tests/test_dual.cpp tests/test_solver.cpp)
target_link_libraries(ddp_tests PRIVATE ddp GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(ddp_tests)
